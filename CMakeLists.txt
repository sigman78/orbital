cmake_minimum_required(VERSION 3.24)
project(Orbital VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORBITAL_BUILD_DEMO "Build the native Vulkan demo" ON)
option(ORBITAL_USE_SLANG "Compile shared-ABI Slang shaders (GLSL remains a comparison path)" ON)
add_library(orbital_scene STATIC src/scene/system.cpp src/scene/geometry.cpp src/procedural/assets.cpp src/app/camera.cpp)
target_include_directories(orbital_scene PUBLIC src)
target_compile_features(orbital_scene PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(orbital_scene PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(orbital_scene PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  foreach(test IN ITEMS system geometry assets camera)
    add_executable(${test}_tests tests/${test}_tests.cpp)
    target_link_libraries(${test}_tests PRIVATE orbital_scene)
    # These tests use assertions as executable invariants, including Release.
    if(MSVC)
      target_compile_options(${test}_tests PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test}_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test}_tests)
  endforeach()
endif()
if(ORBITAL_BUILD_DEMO)
  if(NOT WIN32)
    message(FATAL_ERROR "The desktop demo currently requires Windows. CPU tools can use ORBITAL_BUILD_DEMO=OFF.")
  endif()
  set(NOGRAPHICSAPI_FORCE_CONVENTIONAL_BACKEND ON CACHE BOOL "Use the demo's conventional GLSL descriptor ABI")
  add_subdirectory(third_party/NoGraphicsAPI EXCLUDE_FROM_ALL)
  find_program(SPIRV_VAL_EXECUTABLE NAMES spirv-val HINTS "${CMAKE_SOURCE_DIR}/.tools/Vulkan-ValidationLayers/bin" "$ENV{VULKAN_SDK}/Bin")
  if(ORBITAL_USE_SLANG)
    include(cmake/Slang.cmake)
    foreach(shader IN ITEMS fullscreen surface)
      orbital_compile_slang("shaders/${shader}.vert.spv" "shaders/${shader}.slang" vertexMain vertex)
      list(APPEND ORBITAL_SHADER_OUTPUTS "${CMAKE_BINARY_DIR}/shaders/${shader}.vert.spv")
    endforeach()
    foreach(shader IN ITEMS surface background atmosphere post temporal)
      orbital_compile_slang("shaders/${shader}.frag.spv" "shaders/${shader}.slang" fragmentMain fragment)
      list(APPEND ORBITAL_SHADER_OUTPUTS "${CMAKE_BINARY_DIR}/shaders/${shader}.frag.spv")
    endforeach()
  else()
    find_program(GLSLANG_EXECUTABLE NAMES glslang glslangValidator HINTS "${CMAKE_SOURCE_DIR}/.tools/glslang/bin" "$ENV{VULKAN_SDK}/Bin" REQUIRED)
    include(cmake/Shaders.cmake)
    orbital_shader(fullscreen.vert)
    orbital_shader(surface.vert)
    orbital_shader(surface.frag)
    orbital_shader(background.frag)
    orbital_shader(atmosphere.frag)
    orbital_shader(post.frag)
    orbital_shader(temporal.frag)
  endif()
  add_custom_target(orbital_shaders DEPENDS ${ORBITAL_SHADER_OUTPUTS})
  add_executable(orbital src/app/main.cpp src/app/hud.cpp src/procedural/materials.cpp src/render/renderer.cpp)
  target_link_libraries(orbital PRIVATE orbital_scene NoGraphicsAPI::NoGraphicsAPI user32 gdi32 windowscodecs ole32)
  if(BUILD_TESTING)
    add_executable(materials_tests tests/materials_tests.cpp src/procedural/materials.cpp)
    target_link_libraries(materials_tests PRIVATE orbital_scene windowscodecs ole32)
    target_compile_options(materials_tests PRIVATE /W4 /EHsc /UNDEBUG)
    add_test(NAME materials COMMAND materials_tests "${CMAKE_SOURCE_DIR}/assets/materials")
  endif()
  target_compile_definitions(orbital PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(orbital PRIVATE /W4 /permissive- /EHsc)
  add_dependencies(orbital orbital_shaders)
  add_custom_command(TARGET orbital POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_BINARY_DIR}/shaders" "$<TARGET_FILE_DIR:orbital>/shaders")
  add_custom_target(orbital_assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "${CMAKE_BINARY_DIR}/assets"
    COMMENT "Synchronizing licensed material assets")
  add_dependencies(orbital orbital_assets)
  install(TARGETS orbital RUNTIME DESTINATION .)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/shaders" DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
  install(FILES README.md DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES third_party/NoGraphicsAPI/LICENSE DESTINATION licenses RENAME NoGraphicsAPI-LICENSE.txt)
endif()
